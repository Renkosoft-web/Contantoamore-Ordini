-- 1) Una sola consegna per fascia oraria (opzionale, attivabile da admin) alter table impostazioni_locale add column if not exists una_consegna_per_fascia boolean not null default false; -- 2) Stampante dedicata per singolo prodotto (ha la precedenza su quella della categoria) alter table prodotti add column if not exists stampante_id uuid references stampanti(id) on delete set null; -- 3) Elenco degli orari già occupati per un giorno (usabile anche dal cliente non loggato) create or replace function slot_occupati(p_tenant uuid, p_data date) returns table (ora time) language sql security definer set search_path = public as $$ select o.ora_ritiro from ordini o where o.tenant_id = p_tenant and o.data_ritiro = p_data and coalesce(o.stato, 'nuovo') not in ('annullato', 'cancellato') -- un carrello "in_creazione" blocca la fascia solo per 15 minuti (poi è abbandonato) and (o.stato <> 'in_creazione' or o.creato_il > now() - interval '15 minutes'); $$; grant execute on function slot_occupati(uuid, date) to anon, authenticated; -- 4) Controllo lato database (a prova di due clienti che scelgono lo stesso orario insieme) create or replace function controlla_slot_ordine() returns trigger language plpgsql security definer set search_path = public as $$ declare attivo boolean; begin select una_consegna_per_fascia into attivo from impostazioni_locale where tenant_id = new.tenant_id limit 1; if coalesce(attivo, false) then perform pg_advisory_xact_lock(hashtext(new.tenant_id::text || new.data_ritiro::text || new.ora_ritiro::text)); if exists ( select 1 from ordini o where o.tenant_id = new.tenant_id and o.data_ritiro = new.data_ritiro and o.ora_ritiro = new.ora_ritiro and coalesce(o.stato, 'nuovo') not in ('annullato', 'cancellato') and (o.stato <> 'in_creazione' or o.creato_il > now() - interval '15 minutes') ) then raise exception 'SLOT_OCCUPATO: questo orario è già stato prenotato da un altro cliente'; end if; end if; return new; end; $$; drop trigger if exists trg_controlla_slot_ordine on ordini; create trigger trg_controlla_slot_ordine before insert on ordini for each row execute function controlla_slot_ordine();
